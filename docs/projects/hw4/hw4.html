Figure out how far each point is from our intial centroids. Using broadcasting which is inside the linalg function. Broadcasting will apply the subtraciton across all values in our matrix. and puts it in an array that we can find the minimum of

def assign_clusters(X, centroids):
    distances = np.linalg.norm(X[:, np.newaxis] - centroids, axis=2)
    return np.argmin(distances, axis=1)

import numpy as np

corr_matrix = np.corrcoef(np.column_stack([X_scaled, y_scaled]), rowvar=False)
rxy = corr_matrix[:-1, -1]
rxx = corr_matrix[:-1, :-1]

eig_vals, eig_vecs = np.linalg.eigh(rxx)
lambda_diag = np.diag(np.sqrt(eig_vals))
Z = eig_vecs @ lambda_diag
beta = np.linalg.inv(Z.T @ Z) @ Z.T @ rxy
rel_weights = np.square(Z @ beta)
rel_weights = pd.Series(rel_weights / rel_weights.sum(), index=X.columns)
